# Simulate simple SDE: dX = mu dt + sigma dW
np.random.seed(42)
N=1000
dt = 1/N
X_py = np.zeros(N+1)
for i in range(N):
    dW = np.sqrt(dt) * np.random.normal(0, 1)
    X_py[i+1] = X_py[i] + mu * dt + sigma * dW
